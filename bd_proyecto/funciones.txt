--Consolidación Temporal y Volumetría de Transacciones

--Consulta 1: Determinar cuántos clientes han comprado un producto X en un período determinado
CREATE OR REPLACE FUNCTION f_consolidacion_1(nombre_producto VARCHAR, fecha_inicio DATE, fecha_fin DATE) RETURNS INTEGER AS $$
DECLARE
    resultado INTEGER;
BEGIN
    SELECT COUNT(DISTINCT v.Id_Cliente)
    INTO resultado
    FROM Venta v, Generar3 g3, Producto p
    WHERE g3.Id_Venta = v.Id
    AND p.Id = g3.Id_Producto
    AND LOWER(p.Nombre) = LOWER(nombre_producto)
    AND v.Fecha::DATE BETWEEN fecha_inicio AND fecha_fin;
    RETURN resultado;
END;
$$ LANGUAGE plpgsql;

-- SELECT f_consolidacion_1('Laptop Apple MacBook Air', '2022-01-01', '2024-12-31');

--Consulta 5: Mostrar el total de compras y total pagado del mes actual por cliente X,
--e indique los clientes que han comprado más de 5 veces para asignarles un descuento.

CREATE OR REPLACE FUNCTION f_consolidacion_5(
    p_rut VARCHAR,
    p_fecha_inicio DATE,
    p_fecha_fin DATE
)
RETURNS TABLE (rut VARCHAR, TotalCompras BIGINT, TotalPagado NUMERIC, CalificaDescuento BOOLEAN) AS $$
BEGIN
    RETURN QUERY
    SELECT
        u.rut,
        COUNT(*) AS TotalCompras,
        ROUND(SUM(v.Neto + (v.Neto * v.Iva))) AS TotalPagado,
        COUNT(*) > 5 AS CalificaDescuento
    FROM Usuario u
    JOIN Venta v ON u.rut = v.Id_Cliente
    WHERE u.rut = p_rut
      AND v.Fecha::DATE BETWEEN p_fecha_inicio AND p_fecha_fin
    GROUP BY u.rut;
END;
$$ LANGUAGE plpgsql;

--Consulta 6: Total de ventas del mes por artículo y por vendedor. Totalizando al vendedor

CREATE OR REPLACE FUNCTION f_consolidacion_6(fecha_inicio DATE, fecha_fin DATE)
RETURNS TABLE (RutVendedor VARCHAR, Vendedor VARCHAR, Articulo VARCHAR, TotalVendido NUMERIC) AS $$
BEGIN
    RETURN QUERY
    SELECT resultado.* FROM (
        SELECT v.Id_Vendedor AS RutVendedor, u.Nombre AS Vendedor, p.Nombre AS Articulo, 
               SUM(g3.TotalFinal)::NUMERIC AS TotalVendido
        FROM Venta v, Generar3 g3, Producto p, Usuario u
        WHERE v.Id = g3.Id_Venta
        AND g3.Id_Producto = p.Id
        AND v.Id_Vendedor = u.Rut
        AND v.Fecha::DATE BETWEEN fecha_inicio AND fecha_fin
        GROUP BY v.Id_Vendedor, u.Nombre, p.Nombre
        UNION ALL
        SELECT v.Id_Vendedor, u.Nombre, 'TOTAL VENDEDOR' AS Articulo, 
               SUM(g3.TotalFinal)::NUMERIC AS TotalVendido
        FROM Venta v, Generar3 g3, Producto p, Usuario u
        WHERE v.Id = g3.Id_Venta
        AND g3.Id_Producto = p.Id
        AND v.Id_Vendedor = u.Rut
        AND v.Fecha::DATE BETWEEN fecha_inicio AND fecha_fin
        GROUP BY v.Id_Vendedor, u.Nombre
    ) resultado
    ORDER BY resultado.RutVendedor, (resultado.Articulo = 'TOTAL VENDEDOR'), resultado.Articulo;
END;
$$ LANGUAGE plpgsql;

--Consulta 7: Volumen de ventas: fines de semana vs. días hábiles
CREATE OR REPLACE FUNCTION f_consolidacion_7(fecha_inicio DATE,fecha_fin DATE)
RETURNS TABLE (TipoDia VARCHAR, TotalVentas BIGINT, TotalVendido NUMERIC) AS $$
BEGIN
    RETURN QUERY
    SELECT 'Fin de Semana'::VARCHAR AS TipoDia,
    COUNT(*) AS TotalVentas,
    COALESCE(ROUND(SUM(Neto + (Neto * Iva))), 0) AS TotalVendido
    FROM Venta
    WHERE Fecha::DATE BETWEEN fecha_inicio AND fecha_fin
    AND EXTRACT(DOW FROM Fecha) IN (0, 6)
    
	UNION ALL
    
	SELECT 'Día Hábil'::VARCHAR AS TipoDia,
	COUNT(*) AS TotalVentas,
    COALESCE(ROUND(SUM(Neto + (Neto * Iva))), 0) AS TotalVendido
    FROM Venta
    WHERE Fecha::DATE BETWEEN fecha_inicio AND fecha_fin
    AND EXTRACT(DOW FROM Fecha) NOT IN (0, 6);
END;
$$ LANGUAGE plpgsql;

--Consulta 8: Variación porcentual de ingresos: mes actual vs. mes anterior
CREATE OR REPLACE FUNCTION f_consolidacion_8(p_fecha DATE)
RETURNS TABLE (IngresoMesActual NUMERIC, IngresoMesAnterior NUMERIC, VariacionPorcentual NUMERIC) AS $$
BEGIN
    RETURN QUERY
    SELECT
        ing_actual,
        ing_anterior,
        ROUND(((ing_actual - ing_anterior) / NULLIF(ing_anterior, 0)) * 100, 2) AS VariacionPorcentual
    FROM (
        SELECT
            COALESCE(ROUND((SELECT SUM(Neto + (Neto * Iva)) FROM Venta 
                WHERE Fecha::DATE BETWEEN date_trunc('month', p_fecha)::DATE AND (date_trunc('month', p_fecha) + INTERVAL '1 month - 1 day')::DATE)), 0) AS ing_actual,
            COALESCE(ROUND((SELECT SUM(Neto + (Neto * Iva)) FROM Venta 
                WHERE Fecha::DATE BETWEEN (date_trunc('month', p_fecha) - INTERVAL '1 month')::DATE AND (date_trunc('month', p_fecha) - INTERVAL '1 day')::DATE)), 0) AS ing_anterior
    ) AS ingresos;
END;
$$ LANGUAGE plpgsql;

--Consulta 9: Ventas con descuento aplicado (último trimestre)
CREATE OR REPLACE FUNCTION f_consolidacion_9(anio INTEGER)
RETURNS TABLE (
    Id_Venta INTEGER,
    Fecha TIMESTAMP,
    ValorBruto NUMERIC,
    Cliente VARCHAR,
    PRECIOOFERTA NUMERIC
) AS $$
BEGIN
    RETURN QUERY
    SELECT DISTINCT
        v.Id AS Id_Venta,
        v.Fecha,
        ROUND(v.Neto + (v.Neto * v.Iva)) AS ValorBruto,
        v.Id_Cliente AS Cliente,
        ROUND((v.Neto + (v.Neto * v.Iva)) * (1 - o.preciooferta)) AS PRECIOOFERTA
		FROM Venta v, Generar3 g3, Producto p, oferta o
		WHERE v.Id = g3.Id_Venta
		AND g3.Id_Producto = p.Id
		AND p.Id_Oferta IS NOT NULL
		AND p.Id_Oferta = o.id
		AND EXTRACT(YEAR FROM v.Fecha) = anio
		AND EXTRACT(MONTH FROM v.Fecha) IN (10, 11, 12);
END;
$$ LANGUAGE plpgsql;

RETURNS TABLE (
	EmpresaTransporte VARCHAR,
	PromedioDias NUMERIC
) AS $$

	RETURN QUERY
    SELECT 
        ed.nombre AS EmpresaTransporte, 
        ROUND(AVG(EXTRACT(DAY FROM ((d.datoslogisticos->>'fecha_llegada_real')::TIMESTAMP - (d.datoslogisticos->>'fecha_salida')::TIMESTAMP))), 2) AS PromedioDias

      AND d.datoslogisticos->>'fecha_salida' <> ''
      AND d.datoslogisticos->>'fecha_llegada_real' <> ''

    ORDER BY PromedioDias;

    Rut varchar,
    Nombre varchar
) AS $$
BEGIN
    RETURN QUERY

    SELECT DISTINCT
        u.Rut,
        u.Nombre
    FROM Usuario u, Ofrecer1 o, (
        SELECT e.Id_Producto

    WHERE u.Rut = o.Id_Usuario

    FROM Usuario u, Ofrecer2 o, (
        SELECT e.Id_Servicio
        FROM Escribir2 e, Calificacion c
        WHERE e.Id_Calificacion = c.Id
        GROUP BY e.Id_Servicio
        HAVING AVG(c.Calificacion) < 3
    ) s
    WHERE u.Rut = o.Id_Usuario
      AND o.Id_Servicio = s.Id_Servicio;

END;
$$ LANGUAGE plpgsql;

    Rut varchar,
    Nombre varchar
) AS $$
BEGIN
    RETURN QUERY

    SELECT
        u.Rut,
        u.Nombre
    FROM Usuario u
    WHERE EXISTS (
        -- Verifica que el usuario sea un emprendedor que ofrece servicios
        SELECT 1
        FROM Ofrecer2 o
        WHERE o.Id_Usuario = u.Rut
    )
    AND NOT EXISTS (
        -- Verifica que no tenga calificaciones en los últimos 6 meses
        SELECT 1
        FROM Calificacion c, Escribir2 e, Ofrecer2 o
        WHERE c.Id = e.Id_Calificacion
          AND e.Id_Servicio = o.Id_Servicio
          AND o.Id_Usuario = u.Rut
          AND c.Fecha >= CURRENT_TIMESTAMP - INTERVAL '6 months'
    );

END;
$$ LANGUAGE plpgsql;

-- Funcion 9
CREATE OR REPLACE FUNCTION f_fidelizacion_9(varchar)
RETURNS INTERVAL AS $$
DECLARE
    idvendedor ALIAS FOR $1;
    tiempo_promedio INTERVAL;
BEGIN
    SELECT AVG(r.FechaRespuesta - r.FechaCreacion)
    INTO tiempo_promedio
    FROM (
        -- Preguntas asociadas a productos
        SELECT
            o1.Id_Usuario AS RutVendedor,
            p.FechaCreacion,
            p.FechaRespuesta
        FROM Ofrecer1 o1, Hacer1 h1, Preguntas p
        WHERE o1.Id_Producto = h1.Id_Producto
          AND h1.Id_Pregunta = p.Id
          AND p.FechaRespuesta IS NOT NULL

        UNION ALL

        -- Preguntas asociadas a servicios
        SELECT
            o2.Id_Usuario AS RutVendedor,
            p.FechaCreacion,
            p.FechaRespuesta
        FROM Ofrecer2 o2, Hacer2 h2, Preguntas p
        WHERE o2.Id_Servicio = h2.Id_Servicio
          AND h2.Id_Pregunta = p.Id
          AND p.FechaRespuesta IS NOT NULL
    ) r
    WHERE r.RutVendedor = idvendedor;

    RETURN tiempo_promedio;
END;
$$ LANGUAGE plpgsql;

-- Consulta 3: Calcular el peso promedio de los productos disponibles en inventario, accediendo y operando directamente 
-- el nodo numérico correspondiente dentro de su estructura JSON.
CREATE OR REPLACE FUNCTION f_json_3()
RETURNS TABLE (
    Id_Inventario INTEGER,
    Peso_Promedio NUMERIC
)
AS $$
BEGIN
    RETURN QUERY
    SELECT 
        i.Id AS Id_Inventario,
        (
            SELECT AVG((p.Caracteristicas->>'peso')::numeric)
            FROM Producto p
            WHERE p.Id IN (
                SELECT a.Id_Producto
                FROM Almacenar a
                WHERE a.Id_Inventario = i.Id
            )
        ) AS Peso_Promedio
    FROM Inventario i
    WHERE EXISTS (
        SELECT 1
        FROM Almacenar a
        WHERE a.Id_Inventario = i.Id
          AND EXISTS (
              SELECT 1
              FROM Producto p
              WHERE p.Id = a.Id_Producto
                AND p.Caracteristicas::jsonb ? 'peso'
          )
    );
END;
$$ LANGUAGE plpgsql;

-- Consulta 7: Formular una división relacional que identifique a los clientes que han adquirido al 
-- menos un producto de cada categoría principal disponible en el sistema.

CREATE OR REPLACE FUNCTION f_exclusion_7(
    p_id_categoria_principal INTEGER,
    p_solo_cumplen           BOOLEAN
)
RETURNS TABLE (
    rut_cliente          TEXT,
    nombre_cliente       TEXT,
    categorias_compradas BIGINT,
    total_categorias     BIGINT,
    cumple               BOOLEAN
)
AS $$
DECLARE
    v_total BIGINT;
BEGIN
    IF p_id_categoria_principal IS NULL THEN
        RAISE EXCEPTION 'p_id_categoria_principal no puede ser NULL';
    END IF;
    IF p_solo_cumplen IS NULL THEN
        RAISE EXCEPTION 'p_solo_cumplen no puede ser NULL';
    END IF;

    SELECT COUNT(*) INTO v_total
    FROM Categoria cat
    WHERE cat.Id_Categoria = p_id_categoria_principal;

    RETURN QUERY
    SELECT
        u.Rut::TEXT,
        u.Nombre::TEXT,
        COUNT(DISTINCT c.Id),
        v_total,
        (COUNT(DISTINCT c.Id) = v_total)
    FROM Usuario u, Venta v, Generar3 g3, Producto p, Pertenecer1 p1, Categoria c
    WHERE u.Rut = v.Id_Cliente
      AND v.Id = g3.Id_Venta
      AND g3.Id_Producto = p.Id
      AND p.Id = p1.Id_Producto
      AND p1.Id_Categoria = c.Id
      AND c.Id_Categoria = p_id_categoria_principal
    GROUP BY u.Rut, u.Nombre
    HAVING NOT p_solo_cumplen OR COUNT(DISTINCT c.Id) = v_total
    ORDER BY COUNT(DISTINCT c.Id) DESC, u.Nombre;

    IF NOT FOUND THEN
        RETURN QUERY
        SELECT NULL::TEXT, NULL::TEXT, NULL::BIGINT, v_total, FALSE;
    END IF;
END;
$$ LANGUAGE plpgsql;

-- Consulta 8: Seleccionar los servicios profesionales que son ofrecidos en múltiples regiones 
-- geográficas de manera simultánea, agrupando por identificador y filtrando con HAVING.
CREATE OR REPLACE FUNCTION f_exclusion_8(
    p_id_categoria INTEGER,
    p_min_regiones INTEGER
)
RETURNS TABLE (
    id_servicio     BIGINT,
    nombre_servicio TEXT,
    regiones        BIGINT
)
AS $$
BEGIN
    IF p_id_categoria IS NULL THEN
        RAISE EXCEPTION 'p_id_categoria no puede ser NULL';
    END IF;
    IF p_min_regiones IS NULL OR p_min_regiones < 1 THEN
        RAISE EXCEPTION 'p_min_regiones debe ser un entero mayor o igual a 1 (recibido: %)', p_min_regiones;
    END IF;

    RETURN QUERY
    SELECT
        s.Id::BIGINT,

        COUNT(DISTINCT r.Id)
    FROM Servicio s, Ofrecer2 of2, Usuario u, Registrar2 r2, Direccion d, Comuna c,

      AND (cat.Id = p_id_categoria OR cat.Id_Categoria = p_id_categoria)
    GROUP BY s.Id, s.Nombre
    HAVING COUNT(DISTINCT r.Id) >= p_min_regiones

-- Funcion Inventario 5: Tasa porcentual de devoluciones por emprendedor

CREATE OR REPLACE FUNCTION f_inventario_5()
RETURNS TABLE (
    Emprendedor varchar,
    Ventas bigint,
    Devoluciones bigint,
    PorcentajeDevoluciones numeric
) AS $$
BEGIN
    RETURN QUERY
    SELECT
        u.Nombre AS Emprendedor,
        COUNT(DISTINCT v.Id) AS Ventas,
        COUNT(DISTINCT d.Id) AS Devoluciones,
        ROUND(
            COUNT(DISTINCT d.Id)::NUMERIC * 100 
            / NULLIF(COUNT(DISTINCT v.Id), 0),
            2
        ) AS PorcentajeDevoluciones
    FROM Usuario u
    LEFT JOIN Venta v
        ON u.Rut = v.Id_Vendedor
    LEFT JOIN Generar1 g
        ON v.Id = g.Id_Venta
    LEFT JOIN Devolucion d
        ON g.Id_Devolucion = d.Id
       AND d.Estado = 'Entregado'
    GROUP BY u.Nombre
    ORDER BY PorcentajeDevoluciones DESC NULLS LAST;
END;
$$ LANGUAGE plpgsql;

-- Funcion Inventario 6: Productos con stock y variedad de colores por encima de umbrales parametrizados

    VariantesColor bigint,
    InventarioTotal numeric
) AS $$
DECLARE
    min_colores ALIAS FOR $1;
    min_stock ALIAS FOR $2;

    RETURN QUERY

       AND SUM(a.StockActual) > min_stock
    ORDER BY VariantesColor DESC, InventarioTotal DESC;
END;
$$ LANGUAGE plpgsql;

-- Funcion Inventario 8: Tasa de conversión de productos guardados en listas de deseos vs ventas
CREATE OR REPLACE FUNCTION f_inventario_8(integer)
RETURNS TABLE (
    IdProducto integer,
    NombreProducto varchar,
    VecesGuardado bigint,
    VentasRealizadas bigint,
    TasaConversion numeric
) AS $$
DECLARE
    id_prod_param ALIAS FOR $1;
BEGIN

    SELECT
        p.Id AS IdProducto,
        p.Nombre AS NombreProducto,

        ROUND(
            COUNT(DISTINCT g3.Id_Venta)::NUMERIC * 100 / NULLIF(COUNT(DISTINCT c.Id_Usuario), 0),
            2
        ) AS TasaConversion
    FROM Producto p
    LEFT JOIN Crear1 c
        ON p.Id = c.Id_Producto
    LEFT JOIN Generar3 g3
        ON p.Id = g3.Id_Producto
    WHERE (id_prod_param IS NULL OR p.Id = id_prod_param)
    GROUP BY p.Id, p.Nombre;
END;
$$ LANGUAGE plpgsql;

-- Consulta 1: Diseñar una vista integral que consolide la factura fiscal del cliente, desglosando formalmente 
-- el valor neto, el impuesto IVA, los descuentos y el recargo por envío.
CREATE OR REPLACE FUNCTION f_logicas_1(
    p_id_venta INTEGER
)
RETURNS TABLE (
    folio_factura   INTEGER,
    fecha_emision   DATE,
    rut_cliente     TEXT,
    nombre_cliente  TEXT,
    valor_neto      INTEGER,
    tasa_iva        NUMERIC,
    monto_iva       NUMERIC,
    monto_descuento NUMERIC,
    recargo_envio   NUMERIC,
    total_a_pagar   NUMERIC
)
AS $$
BEGIN
    RETURN QUERY
    WITH Detalle_Descuentos_Productos AS (
        SELECT g3.Id_Venta,
               COALESCE(SUM(
                   CASE
                       WHEN o.PrecioOferta IS NOT NULL AND o.PrecioOferta < 1.0 THEN
                           (g3.TotalFinal / (1.0 - o.PrecioOferta)) * o.PrecioOferta
                       ELSE 0
                   END
               ), 0) AS Descuento_Productos
        FROM Generar3 g3
        JOIN Producto p ON g3.Id_Producto = p.Id
        LEFT JOIN Oferta o ON p.Id_Oferta = o.Id AND o.Estado = TRUE
        GROUP BY g3.Id_Venta
    ),
    Detalle_Descuentos_Servicios AS (
        SELECT g4.Id_Venta,
               COALESCE(SUM(
                   CASE
                       WHEN o.PrecioOferta IS NOT NULL AND o.PrecioOferta < 1.0 THEN
                           (g4.TotalFinal / (1.0 - o.PrecioOferta)) * o.PrecioOferta
                       ELSE 0
                   END
               ), 0) AS Descuento_Servicios
        FROM Generar4 g4
        JOIN Servicio s ON g4.Id_Servicio = s.Id
        LEFT JOIN Oferta o ON s.Id_Oferta = o.Id AND o.Estado = TRUE
        GROUP BY g4.Id_Venta
    ),
    Detalle_Despacho AS (
        SELECT g2.Id_Venta,
               SUM(COALESCE(
                   CAST(d.DatosLogisticos->>'costo_envio' AS DECIMAL),
                   (CAST(d.DatosLogisticos->>'ancho_cm' AS DECIMAL) *
                    CAST(d.DatosLogisticos->>'alto_cm' AS DECIMAL) *
                    CAST(d.DatosLogisticos->>'fondo_cm' AS DECIMAL) * 1.5),
                   0
               )) AS Recargo
        FROM Generar2 g2
        JOIN Despacho d ON g2.Id_Despacho = d.Id
        GROUP BY g2.Id_Venta
    )
    SELECT v.Id::INTEGER,
           v.Fecha::DATE,
           u.Rut::TEXT,
           u.Nombre::TEXT,
           v.Neto::INTEGER,
           v.Iva::NUMERIC,
           ROUND(v.Neto * v.Iva, 0)::NUMERIC,
           ROUND(COALESCE(dp.Descuento_Productos, 0) + COALESCE(ds.Descuento_Servicios, 0), 0)::NUMERIC,
           ROUND(COALESCE(dd.Recargo, 0), 0)::NUMERIC,
           ROUND(v.Neto + (v.Neto * v.Iva) + COALESCE(dd.Recargo, 0), 0)::NUMERIC
    FROM Venta v
    JOIN Usuario u ON v.Id_Cliente = u.Rut
    LEFT JOIN Detalle_Descuentos_Productos dp ON v.Id = dp.Id_Venta
    LEFT JOIN Detalle_Descuentos_Servicios ds ON v.Id = ds.Id_Venta
    LEFT JOIN Detalle_Despacho dd ON v.Id = dd.Id_Venta
    WHERE p_id_venta IS NULL OR v.Id = p_id_venta
    ORDER BY v.Id;
END;
$$ LANGUAGE plpgsql;

-- Consulta 3: Formular una vista de rendimiento métrico que categorice a los 
-- emprendedores en rangos consolidados según su facturación trimestral bruta acumulada.
CREATE OR REPLACE FUNCTION f_logicas_3(p_anio INTEGER, id_vendedor_param VARCHAR)
RETURNS TABLE (
    Id_Vendedor VARCHAR,
    Anio DOUBLE PRECISION,
    Trimestre DOUBLE PRECISION,
    Facturacion_Acumulada NUMERIC,
    Categoria_Rendimiento VARCHAR
) AS $$
BEGIN
    RETURN QUERY
    SELECT
        v.Id_Vendedor,
        DATE_PART('year', v.Fecha) AS Anio,
        DATE_PART('quarter', v.Fecha) AS Trimestre,
        SUM(v.Neto)::NUMERIC AS Facturacion_Acumulada,
        (CASE
            WHEN SUM(v.Neto) < 100000 THEN '< 100.000'
            WHEN SUM(v.Neto) >= 100000 AND SUM(v.Neto) < 500000 THEN '[100.000, 500.000['
            WHEN SUM(v.Neto) >= 500000 AND SUM(v.Neto) < 2000000 THEN '[500.000, 2.000.000['
            ELSE '>=2.000.000'
        END)::VARCHAR AS Categoria_Rendimiento
    FROM Venta v
    WHERE (p_anio IS NULL OR DATE_PART('year', v.Fecha) = p_anio)
      AND (id_vendedor_param IS NULL OR v.Id_Vendedor = id_vendedor_param)
    GROUP BY
        v.Id_Vendedor,
        DATE_PART('year', v.Fecha),
        DATE_PART('quarter', v.Fecha);
END;
$$ LANGUAGE plpgsql;

-- Consulta 4: Estructurar un modelo de consulta que reporte el valor histórico consolidado (Life-Time Value) 
-- de cada cliente, integrando todo su volumen de transacciones en la plataforma.

CREATE OR REPLACE FUNCTION f_logicas_4(
    p_min_transacciones INTEGER
)
RETURNS TABLE (
    rut_cliente         TEXT,
    nombre_cliente      TEXT,
    total_transacciones BIGINT,
    ltv                 NUMERIC,
    ticket_promedio     NUMERIC,
    primera_compra      DATE,
    ultima_compra       DATE,
    antiguedad_dias     INTEGER
)
AS $$
BEGIN
    IF p_min_transacciones IS NULL OR p_min_transacciones < 1 THEN
        RAISE EXCEPTION 'p_min_transacciones debe ser un entero mayor o igual a 1 (recibido: %)', p_min_transacciones;
    END IF;

    RETURN QUERY
    WITH DetalleVentas AS (
        SELECT
            v.Id_Cliente,
            v.Id AS IdVenta,
            v.Fecha,
            (SELECT COALESCE(SUM(g3.TotalFinal), 0) FROM Generar3 g3 WHERE g3.Id_Venta = v.Id) +
            (SELECT COALESCE(SUM(g4.TotalFinal), 0) FROM Generar4 g4 WHERE g4.Id_Venta = v.Id) AS Monto
        FROM Venta v
    )
    SELECT
        u.Rut::TEXT,
        u.Nombre::TEXT,
        COUNT(d.IdVenta),
        SUM(d.Monto)::NUMERIC,
        ROUND(AVG(d.Monto)::NUMERIC, 2),
        MIN(d.Fecha)::DATE,
        MAX(d.Fecha)::DATE,
        (MAX(d.Fecha)::DATE - MIN(d.Fecha)::DATE)::INTEGER
    FROM Usuario u, DetalleVentas d
    WHERE u.Rut = d.Id_Cliente
    GROUP BY u.Rut, u.Nombre
    HAVING COUNT(d.IdVenta) >= p_min_transacciones
    ORDER BY SUM(d.Monto) DESC;
END;
$$ LANGUAGE plpgsql;

-- Consulta 5: Crear una vista jerárquica unificada que mapee la estructura completa y dependencias 
-- de todas las categorías y subcategorías mediante lógica recursiva

CREATE OR REPLACE FUNCTION f_logicas_5(
    p_id_raiz INTEGER
)
RETURNS TABLE (
    id_cat        INTEGER,
    nombre_cat    TEXT,
    id_cat_padre  INTEGER,
    nivel         INTEGER
)
AS $$
BEGIN
    RETURN QUERY
    WITH RECURSIVE Jerarquia (j_id, j_nombre, j_padre, j_nivel) AS (
        SELECT
            c.Id::INTEGER,
            c.Nombre::TEXT,
            c.Id_Categoria::INTEGER,
            1
        FROM Categoria c
        WHERE (p_id_raiz IS NULL AND c.Id_Categoria IS NULL)
           OR c.Id = p_id_raiz
        UNION ALL
        SELECT
            hijo.Id::INTEGER,
            hijo.Nombre::TEXT,
            hijo.Id_Categoria::INTEGER,
            padre.j_nivel + 1
        FROM Categoria hijo
        JOIN Jerarquia padre ON hijo.Id_Categoria = padre.j_id
    )
    SELECT j.j_id, j.j_nombre, j.j_padre, j.j_nivel
    FROM Jerarquia j
    ORDER BY j.j_nivel, j.j_id;
END;
$$ LANGUAGE plpgsql;

-- Consulta 7: Generar una vista geoespacial que agrupe a las empresas registradas según su 
-- giro comercial y totalice la superficie demográfica de las comunas donde mantienen presencia.

CREATE OR REPLACE FUNCTION f_logicas_7(
    p_giro TEXT
)
RETURNS TABLE (
    giro_empresa         TEXT,
    cantidad_empresas    BIGINT,
    cantidad_comunas     BIGINT,
    superficie_total_km2 NUMERIC
)
AS $$
BEGIN
    RETURN QUERY
    WITH presencia AS (
        SELECT DISTINCT e.Giro AS giro, e.Rut AS rut, co.Id AS id_comuna, co.Superficie AS superficie
        FROM Empresa e
        JOIN Registrar5 reg ON reg.Id_Empresa = e.Rut
        JOIN Direccion d ON d.Id = reg.Id_Direccion
        JOIN Comuna co ON co.Id = d.Id_Comuna
        WHERE p_giro IS NULL OR e.Giro ILIKE '%' || p_giro || '%'
    ),
    emp AS (
        SELECT p.giro, COUNT(DISTINCT p.rut) AS n_empresas
        FROM presencia p
        GROUP BY p.giro
    ),
    com AS (
        SELECT p.giro, p.id_comuna, MAX(p.superficie) AS sup
        FROM presencia p
        GROUP BY p.giro, p.id_comuna
    ),
    com_tot AS (
        SELECT c.giro, COUNT(*) AS n_comunas, SUM(c.sup) AS sup_total
        FROM com c
        GROUP BY c.giro
    )
    SELECT emp.giro::TEXT,
           emp.n_empresas,
           ct.n_comunas,
           ct.sup_total::NUMERIC
    FROM emp
    JOIN com_tot ct ON ct.giro IS NOT DISTINCT FROM emp.giro
    ORDER BY ct.sup_total DESC;
END;
$$ LANGUAGE plpgsql;

-- Consulta 8: Diseñar un panel de monitoreo temporal que muestre el promedio aritmético diario de 
-- nuevos registros de clientes en el sistema a lo largo del último mes calendario.

CREATE OR REPLACE FUNCTION f_logicas_8(
    p_fecha_desde DATE,
    p_fecha_hasta DATE
)
RETURNS TABLE (promedio_diario_nuevos_clientes NUMERIC) AS $$
BEGIN

        RAISE EXCEPTION 'p_fecha_desde y p_fecha_hasta no pueden ser NULL';

-- Consulta 10: Formular una vista general de catálogo unificado que integre la disponibilidad de productos físicos y 
-- la oferta de servicios profesionales en una matriz única de búsqueda indexada.

CREATE OR REPLACE FUNCTION f_logicas_10(
    p_busqueda         TEXT,
    p_solo_disponibles BOOLEAN
)
RETURNS TABLE (
    tipo_bien           TEXT,
    id_bien             INTEGER,
    nombre_bien         TEXT,
    restriccion_edad    INTEGER,
    categorias_bien     TEXT,
    id_oferta_bien      INTEGER,
    fecha_inicio_bien   DATE,
    fecha_fin_bien      DATE,
    stock_disponible    BIGINT,
    disponibilidad_bien TEXT
)
AS $$
BEGIN
    IF p_solo_disponibles IS NULL THEN
        RAISE EXCEPTION 'p_solo_disponibles no puede ser NULL';
    END IF;

    RETURN QUERY
    WITH catalogo AS (
        SELECT 'Producto'::TEXT AS TipoBien,
               p.Id AS IdBien,
               p.Nombre AS Nombre,
               p.RestriccionEdad AS RestriccionEdad,
               (SELECT STRING_AGG(c.Nombre::TEXT, ', ')
                FROM Pertenecer1 pe
                JOIN Categoria c ON c.Id = pe.Id_Categoria
                WHERE pe.Id_Producto = p.Id) AS Categorias,
               p.Id_Oferta AS IdOferta,
               p.FechaInicio AS FechaInicio,
               p.FechaFin AS FechaFin,
               st.Total AS Stock,
               CASE WHEN COALESCE(st.Total, 0) > 0
                    THEN 'Disponible' ELSE 'No disponible' END AS Disponibilidad
        FROM Producto p
        LEFT JOIN LATERAL (
            SELECT SUM(a.StockActual) AS Total
            FROM Almacenar a
            WHERE a.Id_Producto = p.Id
        ) st ON TRUE

        UNION ALL

        SELECT 'Servicio'::TEXT,
               s.Id,
               s.Nombre,
               s.RestriccionEdad,
               (SELECT STRING_AGG(c.Nombre::TEXT, ', ')
                FROM Pertenecer2 pe
                JOIN Categoria c ON c.Id = pe.Id_Categoria
                WHERE pe.Id_Servicio = s.Id),
               s.Id_Oferta,
               s.FechaInicio,
               s.FechaFin,
               NULL::BIGINT,
               CASE WHEN s.FechaFin IS NULL OR s.FechaFin >= CURRENT_DATE
                    THEN 'Disponible' ELSE 'No disponible' END
        FROM Servicio s
    )
    SELECT c.TipoBien,
           c.IdBien::INTEGER,
           c.Nombre::TEXT,
           c.RestriccionEdad::INTEGER,
           c.Categorias,
           c.IdOferta::INTEGER,
           c.FechaInicio::DATE,
           c.FechaFin::DATE,
           c.Stock::BIGINT,
           c.Disponibilidad
    FROM catalogo c
    WHERE (p_busqueda IS NULL OR TRIM(p_busqueda) = ''
           OR to_tsvector('spanish', c.Nombre || ' ' || COALESCE(c.Categorias, ''))
              @@ plainto_tsquery('spanish', p_busqueda))
      AND (NOT p_solo_disponibles OR c.Disponibilidad = 'Disponible')
    ORDER BY c.TipoBien, c.Nombre;
END;
$$ LANGUAGE plpgsql;
