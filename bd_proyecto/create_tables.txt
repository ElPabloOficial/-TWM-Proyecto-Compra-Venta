--CREAR TABLAS
--TABLAS SIN FK

--TABLA CARRITOCOMPRA
CREATE TABLE CarritoCompra (
    Id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    Estado VARCHAR(20) NOT NULL,
    UltimaActualizacion TIMESTAMP NOT NULL,
    CONSTRAINT chk_carrito_fecha CHECK (UltimaActualizacion >= '1995-01-01' AND UltimaActualizacion <= CURRENT_DATE)
);

--TABLA CALIFICACIONRESENA
CREATE TABLE Calificacion (
    Id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    Comentario VARCHAR(200),
    Calificacion INTEGER NOT NULL CHECK (Calificacion >= 0),
    Fecha TIMESTAMP NOT NULL,
    CONSTRAINT chk_calificacion_fecha CHECK (Fecha >= '1995-01-01' AND Fecha <= CURRENT_TIMESTAMP)
);

--TABLA LISTADESEOS
CREATE TABLE ListaDeseos (
    Id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    NombreLista VARCHAR(50) NOT NULL,
    Descripcion VARCHAR(200),
    Visibilidad BOOLEAN NOT NULL
);

--TABLA PREGUNTAS
CREATE TABLE Preguntas (
    Id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    Pregunta VARCHAR(200) NOT NULL,
    FechaCreacion TIMESTAMP NOT NULL,
    Respuesta VARCHAR(200),
    FechaRespuesta TIMESTAMP,
    CONSTRAINT chk_preguntas_fecha CHECK (FechaCreacion >= '1995-01-01' AND FechaCreacion <= CURRENT_TIMESTAMP),
    CONSTRAINT chk_preguntas_fecha_respuesta CHECK (FechaRespuesta IS NULL OR FechaRespuesta >= FechaCreacion)
);

--TABLA COTIZACION
CREATE TABLE Cotizacion (
    Id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    FechaCreacion DATE NOT NULL,
    Estado VARCHAR(20) NOT NULL,
    FechaUltimaActualizacion DATE,                                 
    CONSTRAINT chk_cotizacion_fecha CHECK (FechaCreacion >= '1995-01-01' AND FechaCreacion <= CURRENT_DATE)
);
--TABLA OFERTA
CREATE TABLE Oferta (
    Id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    PrecioOferta DECIMAL(3,2) NOT NULL CHECK(PrecioOferta >= 0),
    Estado BOOLEAN
);

--TABLA USUARIO
CREATE TABLE Usuario (
    Rut VARCHAR(11) PRIMARY KEY,
    Nombre VARCHAR(100) NOT NULL,
    Clave VARCHAR(60) NOT NULL,
    FechaNacimiento DATE NOT NULL,
    FechaCreacion DATE NOT NULL,
    TipoUsuario VARCHAR(20) NOT NULL,
    Descuentos JSON,
    SecionIniciada BOOLEAN NOT NULL DEFAULT FALSE,                  
    CONSTRAINT chk_usuario_fecha_nac CHECK (FechaNacimiento >= '1940-01-01'),
    CONSTRAINT chk_usuario_fecha_creacion CHECK (FechaCreacion >= '1995-01-01' AND FechaCreacion <= CURRENT_DATE)
);

--TABLA DEVOLUCION
CREATE TABLE Devolucion (
    Id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    Motivo VARCHAR(200) NOT NULL,
    FechaSolicitud DATE NOT NULL,
    FechaResolucion DATE,
    Estado VARCHAR(20) NOT NULL,
    TipoDevolucion VARCHAR(30) NOT NULL,
    MontoReintegrado INTEGER CHECK (MontoReintegrado >= 0),
    CONSTRAINT chk_devolucion_fecha_sol CHECK (FechaSolicitud >= '1995-01-01'),
    CONSTRAINT chk_devolucion_fecha_res CHECK (FechaResolucion IS NULL OR FechaResolucion >= '1995-01-01')
);

--TABLA MENSAJE
CREATE TABLE Mensaje (

--TABLA PAGO
CREATE TABLE Pago (
    Id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    MedioPago VARCHAR(20) NOT NULL,
    Ult4DiGitosTarjeta INTEGER,
    CodigoTransferencia INTEGER,
    CONSTRAINT chk_pago_tarjeta CHECK (Ult4DiGitosTarjeta IS NULL OR (Ult4DiGitosTarjeta > 999 AND Ult4DiGitosTarjeta < 10000)),
    CONSTRAINT chk_pago_transferencia CHECK (CodigoTransferencia IS NULL OR CodigoTransferencia > 0)

--TABLAS REGION, PROVINCIA Y COMUNA

--TABLAS CON FK
--TABLA PRODUCTO
CREATE TABLE Producto (
    Id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    Nombre VARCHAR(50) NOT NULL,
    RestriccionEdad BOOLEAN NOT NULL,
    Caracteristicas JSON NOT NULL,
    Id_Oferta INTEGER NULL,
    FechaInicio DATE NULL,
    FechaFin DATE NULL,
    RutaImagen TEXT DEFAULT NULL,
    CONSTRAINT chk_fechas CHECK (FechaInicio < FechaFin),
    CONSTRAINT chk_oferta_fechas CHECK (
        (Id_Oferta IS NULL AND FechaInicio IS NULL AND FechaFin IS NULL) OR
        (Id_Oferta IS NOT NULL AND FechaInicio IS NOT NULL AND FechaFin IS NOT NULL)
    ),
    FOREIGN KEY (Id_Oferta) REFERENCES Oferta(Id) ON DELETE SET NULL ON UPDATE CASCADE,
    CONSTRAINT chk_producto_fecha_inicio CHECK (FechaInicio IS NULL OR FechaInicio >= '1995-01-01' AND FechaInicio <= CURRENT_DATE),
    CONSTRAINT chk_producto_fecha_fin CHECK (FechaFin IS NULL OR FechaFin >= '1995-01-01')
);

--TABLA SERVICIO
-- Se agrega columna Caracteristicas (JSON) para soportar filtros como
-- modalidad remoto/presencial, requisitos, etc. (equivalente al JSON de Producto)
CREATE TABLE Servicio (
    Id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    Nombre VARCHAR(50) NOT NULL,
    RestriccionEdad BOOLEAN NOT NULL,
    Duracion TIME NOT NULL,
    Caracteristicas JSON,
    Id_Oferta INTEGER,
    FechaInicio DATE,
    FechaFin DATE,
    RutaImagen TEXT DEFAULT NULL,
    CONSTRAINT chk_fechas_servicio CHECK (FechaInicio < FechaFin),
    CONSTRAINT chk_oferta_fechas_servicio CHECK (
        (Id_Oferta IS NULL AND FechaInicio IS NULL AND FechaFin IS NULL) OR
        (Id_Oferta IS NOT NULL AND FechaInicio IS NOT NULL AND FechaFin IS NOT NULL)
    ),
    FOREIGN KEY (Id_Oferta) REFERENCES Oferta(Id) ON DELETE NO ACTION ON UPDATE CASCADE,
    CONSTRAINT chk_servicio_fecha_inicio CHECK (FechaInicio IS NULL OR FechaInicio >= '1995-01-01' AND FechaInicio <= CURRENT_DATE),
    CONSTRAINT chk_servicio_fecha_fin CHECK (FechaFin IS NULL OR FechaFin >= '1995-01-01')
);

--TABLA DESPACHO
-- Se agrega columna DatosLogisticos (JSON) con peso, dimensiones
-- (ancho/alto/fondo) y fechas de salida/llegada real para el cálculo
-- de tiempos de tránsito y costos de envío proyectados.
CREATE TABLE Despacho (
    Id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    Estado VARCHAR(20) NOT NULL,
    DatosLogisticos JSON,
    Id_Direccion INTEGER NOT NULL,
    FOREIGN KEY (Id_Direccion) REFERENCES Direccion(Id) ON DELETE NO ACTION ON UPDATE CASCADE,
    CONSTRAINT chk_despacho_estado CHECK (Estado IN ('En Preparación', 'Despachado', 'En reparto', 'Entregado'))
);

--TABLA VENTA
-- Nota: motivo del descuento (10% por más de 5 compras en el mes, o 5% por inactividad).

    Id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    Neto INTEGER NOT NULL CHECK(Neto > 0),
    Iva DECIMAL(3,2) NOT NULL,
    Fecha TIMESTAMP,
    RequiereDespacho BOOLEAN NOT NULL,
    Estado VARCHAR(20) NOT NULL,
    Id_Vendedor VARCHAR(11) NOT NULL,
    Id_Cliente VARCHAR(11) NOT NULL,
    Nota VARCHAR(200),                                -- NUEVO
    FOREIGN KEY (Id_Vendedor) REFERENCES Usuario(Rut) ON DELETE NO ACTION ON UPDATE CASCADE,
    FOREIGN KEY (Id_Cliente) REFERENCES Usuario(Rut) ON DELETE NO ACTION ON UPDATE CASCADE,
    CONSTRAINT chk_venta_fecha CHECK (Fecha >= '1995-01-01' AND Fecha <= CURRENT_TIMESTAMP),
    CONSTRAINT chk_venta_vendedor_cliente CHECK (Id_Cliente <> Id_Vendedor)
);
