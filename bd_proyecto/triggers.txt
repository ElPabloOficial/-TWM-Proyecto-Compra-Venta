--TRIGGER 1: Los clientes que han comprado más de 5 veces en el mismo mes asignarles un descuento
-- del 10%. Agregar un nota/descripción en la compra de esta acción.

CREATE OR REPLACE FUNCTION aplicar_descuento_mensual()
RETURNS TRIGGER
AS $$
DECLARE
    v_fecha       TIMESTAMP := COALESCE(NEW.Fecha, CURRENT_TIMESTAMP);
    v_compras_mes INTEGER;
BEGIN
    SELECT COUNT(*)
    INTO v_compras_mes
    FROM Venta
    WHERE Id_Cliente = NEW.Id_Cliente
      AND Fecha >= date_trunc('month', v_fecha)
      AND Fecha <  date_trunc('month', v_fecha) + INTERVAL '1 month';

    IF v_compras_mes > 5 THEN
        NEW.Neto := ROUND(NEW.Neto * 0.90);
        NEW.Nota := 'Descuento 10% por más de 5 compras en el mismo mes';
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- TRIGGER 2: Calcular dinámicamente el costo de envío proyectado
-- basándose en el volumen (ancho, alto, fondo) de DatosLogisticos.

CREATE OR REPLACE FUNCTION calculo_costo_envio_proyectado()
RETURNS TRIGGER AS $$
DECLARE
    v_ancho NUMERIC;
    v_alto  NUMERIC;
    v_fondo NUMERIC;
    v_costo_proyectado NUMERIC;
BEGIN
    IF NEW.DatosLogisticos IS NOT NULL THEN
        v_ancho := NULLIF((NEW.DatosLogisticos::jsonb)->>'ancho_cm', '')::NUMERIC;
        v_alto  := NULLIF((NEW.DatosLogisticos::jsonb)->>'alto_cm', '')::NUMERIC;
        v_fondo := NULLIF((NEW.DatosLogisticos::jsonb)->>'fondo_cm', '')::NUMERIC;

        IF v_ancho IS NOT NULL AND v_alto IS NOT NULL AND v_fondo IS NOT NULL THEN
            v_costo_proyectado := ROUND((v_ancho * v_alto * v_fondo / 5000.0 * 1555.0), 0);

            NEW.DatosLogisticos := jsonb_set(
                NEW.DatosLogisticos::jsonb,
                '{costo_envio}',
                to_jsonb(v_costo_proyectado),
                TRUE
            )::json;
        END IF;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- TRIGGER 3: Habilitar un campo en productos/servicio que permita ocultar un productos/servicio que
-- no tenga actividad en más de 40 días.
-- Actividad: al menos una venta en los últimos 40 días.

CREATE OR REPLACE FUNCTION fn_ocultar_inactivos() RETURNS TRIGGER AS $$
BEGIN
    UPDATE Producto p
    SET Caracteristicas = jsonb_set(p.Caracteristicas::jsonb, '{visibilidad}', '"Oculto"')::json
    WHERE (p.Caracteristicas->>'visibilidad') = 'Activo'
      AND NOT EXISTS (
          SELECT 1 FROM Generar3 g, Venta v
          WHERE g.Id_Venta = v.Id AND g.Id_Producto = p.Id
            AND v.Fecha >= CURRENT_TIMESTAMP - INTERVAL '40 days');

    UPDATE Servicio s
    SET Caracteristicas = jsonb_set(COALESCE(s.Caracteristicas::jsonb, '{}'::jsonb), '{visibilidad}', '"Oculto"')::json
    WHERE COALESCE(s.Caracteristicas->>'visibilidad', 'Activo') = 'Activo'
      AND NOT EXISTS (
          SELECT 1 FROM Generar4 g, Venta v
          WHERE g.Id_Venta = v.Id AND g.Id_Servicio = s.Id
            AND v.Fecha >= CURRENT_TIMESTAMP - INTERVAL '40 days');

    RETURN NULL;
END;
$$ LANGUAGE plpgsql;

-- TRIGGER 4: Verificar que un producto/servicio solo pueda tener una oferta activa al mismo tiempo, no
--tiene que ser periodo exacto


CREATE OR REPLACE FUNCTION fn_oferta_unica_activa() RETURNS TRIGGER AS $$
DECLARE
    v_oferta_activa BOOLEAN;
BEGIN
    IF NEW.Id_Oferta IS DISTINCT FROM OLD.Id_Oferta AND OLD.Id_Oferta IS NOT NULL THEN
        SELECT (o.Estado IS TRUE AND CURRENT_DATE BETWEEN OLD.FechaInicio AND OLD.FechaFin)
        INTO v_oferta_activa
        FROM Oferta o WHERE o.Id = OLD.Id_Oferta;
 
        IF v_oferta_activa THEN
            RAISE EXCEPTION 'No se puede asignar una nueva oferta: la oferta actual (%) sigue vigente hasta %.',
                OLD.Id_Oferta, OLD.FechaFin;
        END IF;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- TRIGGER 5: Verificar que si un producto/servicio posee oferta vigente, usar este precio, si no, usar el
-- precio original. Usa f_inventario_2(TRUE).

CREATE OR REPLACE FUNCTION fn_calcular_total_generar3()
RETURNS TRIGGER AS $$
DECLARE
    v_precio    NUMERIC;
    v_descuento NUMERIC;
BEGIN
    SELECT (Caracteristicas->>'precio')::NUMERIC
    INTO v_precio
    FROM Producto
    WHERE Id = NEW.Id_Producto;

    IF v_precio IS NULL THEN
        RAISE EXCEPTION 'El producto % no tiene precio definido.', NEW.Id_Producto;
    END IF;

    SELECT PrecioOferta
    INTO v_descuento
    FROM f_inventario_2(TRUE)
    WHERE IdProducto = NEW.Id_Producto
      AND CURRENT_DATE BETWEEN FechaInicio AND FechaFin
    LIMIT 1;

    NEW.TotalFinal := ROUND(v_precio * (1 - COALESCE(v_descuento, 0)) * NEW.Cantidad);

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- TRIGGER 5 (servicios): el precio del servicio está en Valorar
-- (Id_Servicio, Id_Usuario, Precio);

CREATE OR REPLACE FUNCTION fn_calcular_total_generar4()
RETURNS TRIGGER AS $$
DECLARE
    v_vendedor  VARCHAR(11);
    v_precio    NUMERIC;
    v_descuento NUMERIC;
BEGIN
    SELECT Id_Vendedor INTO v_vendedor FROM Venta WHERE Id = NEW.Id_Venta;

    IF v_vendedor IS NULL THEN
        RAISE EXCEPTION 'La venta % no existe.', NEW.Id_Venta;
    END IF;

    SELECT Precio INTO v_precio
    FROM Valorar
    WHERE Id_Servicio = NEW.Id_Servicio AND Id_Usuario = v_vendedor;

    IF v_precio IS NULL THEN
        RAISE EXCEPTION 'El servicio % no tiene precio valorado por el vendedor %.',
            NEW.Id_Servicio, v_vendedor;
    END IF;

    SELECT o.PrecioOferta INTO v_descuento
    FROM Servicio s, Oferta o
    WHERE s.Id = NEW.Id_Servicio
      AND s.Id_Oferta = o.Id
      AND o.Estado IS TRUE
      AND CURRENT_DATE BETWEEN s.FechaInicio AND s.FechaFin;

    NEW.TotalFinal := ROUND(v_precio * (1 - COALESCE(v_descuento, 0)) * NEW.Cantidad);

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- TRIGGER 6: Impedir la contratación de servicios con barrera etaria
-- por parte de clientes que no cumplen la edad mínima.

CREATE OR REPLACE FUNCTION fn_validar_edad_servicio() RETURNS TRIGGER AS $$
DECLARE
    v_restriccion BOOLEAN;
    v_edad INTEGER;
    v_id_cliente VARCHAR(11);
    v_edad_minima CONSTANT INTEGER := 18;
BEGIN
    SELECT RestriccionEdad INTO v_restriccion FROM Servicio WHERE Id = NEW.Id_Servicio;
    SELECT Id_Cliente INTO v_id_cliente FROM Venta WHERE Id = NEW.Id_Venta;
    SELECT DATE_PART('year', AGE(FechaNacimiento)) INTO v_edad FROM Usuario WHERE Rut = v_id_cliente;

    IF v_restriccion = TRUE AND (v_edad IS NULL OR v_edad < v_edad_minima) THEN
        RAISE EXCEPTION 'El cliente % (edad %) no cumple la edad mínima (%) para contratar el servicio %.',
            v_id_cliente, v_edad, v_edad_minima, NEW.Id_Servicio;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- TRIGGER 7: Detectar direcciones de correo electrónico registradas que presenten anomalías
--estructurales o dominios de red inválidos, utilizando expresiones regulares o LIKE.


CREATE OR REPLACE FUNCTION validar_correo_trigger()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.Correo IS NULL
       OR NEW.Correo NOT LIKE '%_@_%.__%'
       OR NEW.Correo LIKE '%@%@%'
       OR NEW.Correo LIKE '@%'
       OR NEW.Correo LIKE '%@'
       OR NEW.Correo LIKE '%..%'
       OR NEW.Correo LIKE '% %'
       OR NEW.Correo LIKE '.%'
       OR NEW.Correo LIKE '%.'
       OR NEW.Correo LIKE '%.@%'
       OR NEW.Correo LIKE '%@.%'
    THEN
        RAISE EXCEPTION
            'Correo electrónico inválido: %',
            NEW.Correo;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- TRIGGER 9:Cada compra que se concrete de un producto descontar el stock

CREATE OR REPLACE FUNCTION fn_venta_concretada(p_estado VARCHAR) RETURNS BOOLEAN AS $$
BEGIN
    RETURN COALESCE(p_estado IN ('Pagado', 'Entregado', 'Completada'), FALSE);
END;
$$ LANGUAGE plpgsql;


CREATE OR REPLACE FUNCTION fn_descontar_stock_producto(p_id_producto INTEGER, p_cantidad INTEGER)
RETURNS VOID AS $$
DECLARE
    r           Almacenar%ROWTYPE;
    v_restante  INTEGER := p_cantidad;
    v_tomar     INTEGER;
    v_total     INTEGER;
BEGIN
    SELECT COALESCE(SUM(StockActual), 0) INTO v_total
    FROM Almacenar WHERE Id_Producto = p_id_producto;

    IF v_total < p_cantidad THEN
        RAISE EXCEPTION 'Stock insuficiente para el producto %: se piden % unidades y hay % en total.',
            p_id_producto, p_cantidad, v_total;
    END IF;

    FOR r IN SELECT *
             FROM Almacenar
             WHERE Id_Producto = p_id_producto AND StockActual > 0
             ORDER BY StockActual DESC
             FOR UPDATE
    LOOP
        EXIT WHEN v_restante <= 0;
        v_tomar := LEAST(r.StockActual, v_restante);

        UPDATE Almacenar
        SET StockActual = StockActual - v_tomar
        WHERE Id_Inventario = r.Id_Inventario AND Id_Producto = p_id_producto;

        v_restante := v_restante - v_tomar;
    END LOOP;

    IF v_restante > 0 THEN
        RAISE EXCEPTION 'Stock insuficiente para el producto % (faltaron % unidades).',
            p_id_producto, v_restante;
    END IF;
END;
$$ LANGUAGE plpgsql;

-- TRIGGER 10: Cada devolución que se concrete de un producto aumentar el stock.

CREATE OR REPLACE FUNCTION fn_aumentar_stock_devolucion() RETURNS TRIGGER AS $$
DECLARE
    r     Generar1%ROWTYPE;
    v_alm Almacenar%ROWTYPE;
BEGIN
    IF NEW.Estado = 'Entregado' AND OLD.Estado IS DISTINCT FROM 'Entregado' THEN
        FOR r IN SELECT * FROM Generar1 WHERE Id_Devolucion = NEW.Id LOOP
            SELECT * INTO v_alm
            FROM Almacenar
            WHERE Id_Producto = r.Id_Producto
            ORDER BY StockActual ASC
            LIMIT 1
            FOR UPDATE;

            IF FOUND THEN
                UPDATE Almacenar
                SET StockActual = StockActual + r.Cantidad
                WHERE Id_Inventario = v_alm.Id_Inventario
                  AND Id_Producto   = v_alm.Id_Producto;
            ELSE
                RAISE WARNING 'No existe registro de stock para el producto % al procesar la devolución %.',
                    r.Id_Producto, NEW.Id;
            END IF;
        END LOOP;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- TRIGGER 13: Validar la consistencia numérica, confirmando que el importe total abonado coincide con la suma aritmética de brutos e IVA, restando el descuento

CREATE OR REPLACE FUNCTION fn_validar_consistencia_venta()
RETURNS TRIGGER AS $$
DECLARE
    v_id_venta        INTEGER;
    v_neto            INTEGER;
    v_iva             NUMERIC;
    v_requiere        BOOLEAN;
    v_total_productos NUMERIC;
    v_total_servicios NUMERIC;
    v_envio           NUMERIC := 0;
    v_esperado        NUMERIC;
    v_nota            VARCHAR(200);
    v_factor          NUMERIC := 1;
BEGIN
    IF TG_TABLE_NAME = 'venta' THEN
        v_id_venta := NEW.Id;
    ELSE
        v_id_venta := NEW.Id_Venta;
    END IF;

    SELECT Neto, Iva, RequiereDespacho, Nota INTO v_neto, v_iva, v_requiere, v_nota
    FROM Venta
    WHERE Id = v_id_venta;

    IF NOT FOUND THEN
        RETURN NULL;
    END IF;

    IF v_iva <> 0.19 THEN
        RAISE EXCEPTION 'IVA inválido en Venta %: se registró % y debe ser 0.19 (19%%).',
            v_id_venta, v_iva;
    END IF;

    SELECT COALESCE(SUM(TotalFinal), 0) INTO v_total_productos
    FROM Generar3 WHERE Id_Venta = v_id_venta;

    SELECT COALESCE(SUM(TotalFinal), 0) INTO v_total_servicios
    FROM Generar4 WHERE Id_Venta = v_id_venta;

    IF v_requiere THEN
        SELECT COALESCE(SUM(NULLIF(d.DatosLogisticos->>'costo_envio', '')::NUMERIC), 0)
        INTO v_envio
        FROM Generar2 g2
        JOIN Despacho d ON d.Id = g2.Id_Despacho
        WHERE g2.Id_Venta = v_id_venta;
    END IF;

    -- Si la venta tiene descuento (lo indica Venta.Nota), el Neto debe ser la
    -- suma de los detalles con ese descuento aplicado.
    v_factor := CASE WHEN v_nota LIKE 'Descuento 10%' THEN 0.90
                     WHEN v_nota LIKE 'Descuento 5%'  THEN 0.95
                     ELSE 1 END;

    v_esperado := ROUND((v_total_productos + v_total_servicios + v_envio) * v_factor);

    IF v_esperado <> v_neto THEN
        RAISE EXCEPTION
            'Inconsistencia numérica en Venta %: Neto registrado = %, esperado = % (productos % + servicios % + envío %), bruto esperado con IVA = %.',
            v_id_venta, v_neto, v_esperado, v_total_productos, v_total_servicios, v_envio,
            ROUND(v_esperado * (1 + v_iva));
    END IF;

    RETURN NULL;
END;
$$ LANGUAGE plpgsql;

--Trigger 14: Validación y Corrección del Dígito Verificador del RUT: Un trigger que se active antes de
--insertar o actualizar un Cliente o Emprendedor. Validará el dígito verificador del RUT; si es
--incorrecto, lo corregirá y generará una advertencia (NOTICE)

CREATE OR REPLACE FUNCTION fn_calcular_dv(numero BIGINT) RETURNS VARCHAR(1) AS $$
DECLARE
    suma INTEGER := 0;
    factor INTEGER := 2;
    resto INTEGER := 0 ;
BEGIN
    WHILE numero > 0 LOOP
        suma := suma + (numero % 10) * factor;
        numero := numero / 10;
        factor := factor + 1;
        IF factor > 7 THEN
            factor := 2;
        END IF;
    END LOOP;

    resto := 11 - (suma % 11);

    IF resto = 11 THEN RETURN '0';
    ELSIF resto = 10 THEN RETURN 'K';
    ELSE RETURN resto::TEXT;
    END IF;
END;
$$ LANGUAGE plpgsql;


CREATE OR REPLACE FUNCTION fn_validar_corregir_rut() RETURNS TRIGGER AS $$
DECLARE
    numero_rut BIGINT;
    dv_ingresado VARCHAR(1);
    dv_correcto VARCHAR(1);
BEGIN
    numero_rut := SPLIT_PART(NEW.Rut, '-', 1)::BIGINT;
    dv_ingresado := UPPER(SPLIT_PART(NEW.Rut, '-', 2));
    dv_correcto := fn_calcular_dv(numero_rut);

    IF dv_ingresado <> dv_correcto THEN
        RAISE NOTICE 'RUT % tenía el DV incorrecto (%); se corrigió a %-%.',
            NEW.Rut, dv_ingresado, numero_rut, dv_correcto;
        NEW.Rut := numero_rut || '-' || dv_correcto;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION fn_evitar_eliminar_cliente() RETURNS TRIGGER AS $$
DECLARE
    v_pendientes INTEGER;
BEGIN
    SELECT COUNT(*) INTO v_pendientes
    FROM Venta v
    LEFT JOIN Generar2 g2 ON g2.Id_Venta = v.Id
    LEFT JOIN Despacho d ON d.Id = g2.Id_Despacho
    WHERE v.Id_Cliente = OLD.Rut
      AND v.Estado <> 'Anulado'
      AND (
            v.Estado IN ('Pendiente','En proceso','Pagado')

          );

        RAISE EXCEPTION 'No se puede eliminar al cliente %: tiene compras sin entregar o dentro de los últimos 60 días.', OLD.Rut;
    END IF;
    RETURN OLD;
END;
$$ LANGUAGE plpgsql;

-- TRIGGER 18:  Impedir Modificación de Facturas/Boletas Emitidas: Prevenga la modificación de cualquier 
--campo de una Factura/Boleta una vez que ha sido marcada como “Pagada” y/o su estado esta como “Entregado”

CREATE OR REPLACE FUNCTION fn_bloquear_venta_pagada() RETURNS TRIGGER AS $$
BEGIN
    IF OLD.Estado = 'Entregado' THEN
        RAISE EXCEPTION 'No se puede modificar la Venta % porque ya está marcada como Entregado.', OLD.Id;
    ELSIF OLD.Estado = 'Pagado' THEN
        IF (NEW.Id, NEW.Neto, NEW.Iva, NEW.Fecha, NEW.RequiereDespacho, NEW.Id_Vendedor, NEW.Id_Cliente, NEW.Nota)
           IS DISTINCT FROM
           (OLD.Id, OLD.Neto, OLD.Iva, OLD.Fecha, OLD.RequiereDespacho, OLD.Id_Vendedor, OLD.Id_Cliente, OLD.Nota)
           OR NEW.Estado NOT IN ('Pagado', 'Entregado')
        THEN
            RAISE EXCEPTION 'No se puede modificar la Venta % porque ya está Pagada (solo puede pasar a Entregado).', OLD.Id;
        END IF;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- TRIGGER 19: Los clientes pueden seleccionar uno o varios productos para solicitar una
-- cotización, esta debe almacenar la fecha, productos, valores y tendrá una duración de 15 días,
-- después de eso los precios deberán ajustarse automáticamente cada semana y después de 45 días
-- debe ser eliminada la cotización.

CREATE OR REPLACE FUNCTION gestionar_cotizaciones()
RETURNS TRIGGER
AS $$
BEGIN

    IF OLD.SecionIniciada IS DISTINCT FROM NEW.SecionIniciada
       AND NEW.SecionIniciada = TRUE THEN

        -- 1. Eliminar cotizaciones que cumplieron 45 días
        DELETE FROM Cotizacion c
        WHERE c.FechaCreacion <= CURRENT_DATE - 45
          AND (
              EXISTS (
                  SELECT 1
                  FROM Solicitar1 s1
                  WHERE s1.Id_Cotizacion = c.Id
                    AND s1.Id_Usuario = NEW.Rut
              )
              OR EXISTS (
                  SELECT 1
                  FROM Solicitar2 s2
                  WHERE s2.Id_Cotizacion = c.Id
                    AND s2.Id_Usuario = NEW.Rut
              )
          );

        UPDATE Solicitar1 s
        SET Precio = (p.Caracteristicas->>'precio')::INTEGER
        FROM Producto p, Cotizacion c
        WHERE s.Id_Producto = p.Id
          AND s.Id_Cotizacion = c.Id
          AND s.Id_Usuario = NEW.Rut
          AND c.FechaCreacion <= CURRENT_DATE - 15
          AND CURRENT_DATE >=
              COALESCE(c.FechaUltimaActualizacion, c.FechaCreacion) + 7
          AND c.FechaCreacion > CURRENT_DATE - 45
          AND p.Caracteristicas->>'precio' IS NOT NULL;

        UPDATE Solicitar2 s
        SET Precio = v.Precio
        FROM Valorar v, Cotizacion c
        WHERE s.Id_Servicio = v.Id_Servicio
          AND s.Id_Oferente = v.Id_Usuario
          AND s.Id_Cotizacion = c.Id
          AND s.Id_Usuario = NEW.Rut
          AND c.FechaCreacion <= CURRENT_DATE - 15
          AND CURRENT_DATE >=
              COALESCE(c.FechaUltimaActualizacion, c.FechaCreacion) + 7
          AND c.FechaCreacion > CURRENT_DATE - 45;

        UPDATE Cotizacion c
        SET FechaUltimaActualizacion = CURRENT_DATE
        WHERE c.FechaCreacion <= CURRENT_DATE - 15
          AND CURRENT_DATE >=
              COALESCE(c.FechaUltimaActualizacion, c.FechaCreacion) + 7
          AND c.FechaCreacion > CURRENT_DATE - 45
          AND (
              EXISTS (
                  SELECT 1
                  FROM Solicitar1 s1
                  WHERE s1.Id_Cotizacion = c.Id
                    AND s1.Id_Usuario = NEW.Rut
              )
              OR EXISTS (
                  SELECT 1
                  FROM Solicitar2 s2
                  WHERE s2.Id_Cotizacion = c.Id
                    AND s2.Id_Usuario = NEW.Rut
              )
          );

    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_gestionar_cotizaciones
AFTER UPDATE OF SecionIniciada
ON Usuario
FOR EACH ROW
WHEN (
    OLD.SecionIniciada IS DISTINCT FROM NEW.SecionIniciada
    AND NEW.SecionIniciada = TRUE
)
EXECUTE FUNCTION gestionar_cotizaciones();

--TRIGGER 21: Validación Masiva y Corrección de RUTs: Desarrolla una función que itere sobre todos los
--clientes y personal en la base de datos para validar el dígito verificador de sus RUTs. En
--caso de ser incorrecto enviar una una advertencia.
--Requiere fn_calcular_dv (Trigger 14)

CREATE OR REPLACE FUNCTION fn_validar_ruts_masivo() RETURNS INTEGER AS $$
DECLARE
    r                 Usuario%ROWTYPE;
    numero_rut        BIGINT;
    dv_ingresado      VARCHAR(1);
    dv_correcto       VARCHAR(1);
    total_revisados   INTEGER := 0;
    total_incorrectos INTEGER := 0;
BEGIN
    FOR r IN SELECT * FROM Usuario LOOP
        total_revisados := total_revisados + 1;
        numero_rut   := SPLIT_PART(r.Rut, '-', 1)::BIGINT;
        dv_ingresado := UPPER(SPLIT_PART(r.Rut, '-', 2));
        dv_correcto  := fn_calcular_dv(numero_rut);

        IF dv_ingresado <> dv_correcto THEN
            RAISE WARNING 'RUT inválido: % (dígito correcto: %)', r.Rut, dv_correcto;
            total_incorrectos := total_incorrectos + 1;
        END IF;
    END LOOP;

    RAISE NOTICE 'Revisión terminada: % RUT revisados, % con problemas.',
        total_revisados, total_incorrectos;

    RETURN total_incorrectos;
END;
$$ LANGUAGE plpgsql;

-- Trigger 22:Los clientes que no hayan tenido actividad (compras) en los últimos 6 meses asignarles
-- un 5% dedescuento en la próxima compra.

CREATE OR REPLACE FUNCTION revisar_descuento_carrito()
RETURNS TRIGGER
AS $$
DECLARE
    v_id_usuario    VARCHAR(11);
    v_tiene_marca   BOOLEAN;
    v_ultima_compra TIMESTAMP;
BEGIN
    SELECT c3.Id_Usuario
    INTO v_id_usuario
    FROM Crear3 AS c3
    WHERE c3.Id_Carrito = NEW.Id
    LIMIT 1;

    IF v_id_usuario IS NULL THEN
        SELECT c4.Id_Usuario
        INTO v_id_usuario
        FROM Crear4 AS c4
        WHERE c4.Id_Carrito = NEW.Id
        LIMIT 1;
    END IF;

    IF v_id_usuario IS NULL THEN
        RETURN NEW;
    END IF;

    SELECT (u.Descuentos->>'descuento5%') IS NOT NULL
    INTO v_tiene_marca
    FROM Usuario AS u
    WHERE u.Rut = v_id_usuario;

    IF COALESCE(v_tiene_marca, FALSE) THEN
        RETURN NEW;
    END IF;

    SELECT MAX(v.Fecha)
    INTO v_ultima_compra
    FROM Venta AS v
    WHERE v.Id_Cliente = v_id_usuario;

    IF v_ultima_compra IS NOT NULL
       AND v_ultima_compra <= CURRENT_TIMESTAMP - INTERVAL '6 months' THEN
        UPDATE Usuario AS u
        SET Descuentos = (COALESCE(u.Descuentos::jsonb, '{}'::jsonb)
                          || jsonb_build_object('descuento5%', CURRENT_DATE))::json
        WHERE u.Rut = v_id_usuario;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

AFTER UPDATE ON CarritoCompra
FOR EACH ROW

--Trigger 23:La duración del descuento de 5% que se asigne a un cliente debe durar solo 45 días.

CREATE OR REPLACE FUNCTION aplicar_descuento_inactividad()
RETURNS TRIGGER
AS $$
DECLARE
    v_fecha TIMESTAMP := COALESCE(NEW.Fecha, CURRENT_TIMESTAMP);
    v_marca DATE;
BEGIN
    SELECT (Descuentos->>'descuento5%')::DATE
    INTO v_marca
    FROM Usuario
    WHERE Rut = NEW.Id_Cliente;

    IF v_marca IS NULL THEN
        RETURN NEW;
    END IF;

    -- vencida (más de 45 días): se elimina y no se aplica
    IF v_fecha::DATE > v_marca + 45 THEN
        UPDATE Usuario
        SET Descuentos = (Descuentos::jsonb - 'descuento5%')::json
        WHERE Rut = NEW.Id_Cliente;
        RETURN NEW;
    END IF;

    -- si el T1 ya aplicó el 10%, no se acumula
    IF NEW.Nota IS NOT NULL THEN
        RETURN NEW;
    END IF;

    NEW.Neto := ROUND(NEW.Neto * 0.95);
    NEW.Nota := 'Descuento 5% por inactividad de 6 meses (vigencia 45 días)';

    UPDATE Usuario
    SET Descuentos = (Descuentos::jsonb - 'descuento5%')::json
    WHERE Rut = NEW.Id_Cliente;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;
